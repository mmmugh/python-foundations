            indent = "  " * depth
            print(f"{indent}factorial({n}) called")
        
            if n <= 1:
                print(f"{indent}base case, returning 1")
                return 1
        
            result = n * factorial_traced(n - 1, depth + 1)
            print(f"{indent}returning {n} * factorial({n - 1}) = {result}")
            return result
